// ========================================================= // Supabase Edge Function: free-research // // 無料版（Amazonリサーチだけ）の中継役。 // // ・ライセンスキーは使わない。誰でも開ける。 // ・Keepaを叩くのは「お客さん自身のKeepa APIキー」。 // こちらのトークンは1つも減らない。 // ・Keepaキーは保存しない。ハッシュだけを回数の数え札に使う。 // ・1日20回まで（解析の回数） // ・解析するたびに、型番とメーカー名を辞書に貯める。 // 価格やランクは貯めない（Keepaの規約に触れるため）。 // // 🌟 速さのための作り // ・kind:"auto" … 検索から商品取得までを、サーバーの中で続けてやる。 // 以前はブラウザが「検索」で1往復、「商品」でもう1往復していた。 // バーコードを読んだときは必ず2往復になっていたのを1往復に減らす。 // ・ping … 画面を開いた瞬間に軽く1回叩いてもらい、関数を温めておく。 // Edge Functionはしばらく呼ばれていないと起動に1〜3秒かかる。 // ・offers は付けない。出品者一覧をその場で取りに行くと遅く、 // トークンも余計に減る。無料版の画面では使っていない。 // ========================================================= import { serve } from "https://deno.land/std@0.203.0/http/server.ts"; import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!; const SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const supabase = createClient(SUPABASE_URL, SERVICE_ROLE_KEY); // 1日に解析できる回数 const FREE_DAILY_LIMIT = 20; // ここから来た通信だけ受け付ける。空にすると誰からでも受ける。 const ALLOWED_ORIGINS = [ "https://searchhubpro.github.io", ]; const CORS_HEADERS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; function json(body: unknown, status = 200) { return new Response(JSON.stringify(body), { status, headers: { "Content-Type": "application/json", ...CORS_HEADERS }, }); } // 日本時間で「今日」を出す function todayJST(): string { const d = new Date(Date.now() + 9 * 60 * 60 * 1000); return d.toISOString().slice(0, 10); } // Keepaキーを、戻せない形に潰す async function sha256(text: string): Promise { const buf = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(text)); return Array.from(new Uint8Array(buf)).map((b) => b.toString(16).padStart(2, "0")).join(""); } // 自分のページから来た通信かどうか function originAllowed(req: Request): boolean { if (ALLOWED_ORIGINS.length === 0) return true; const origin = req.headers.get("origin") || ""; if (origin && ALLOWED_ORIGINS.includes(origin)) return true; // スマホのホーム画面から開いた場合など、originが付かないことがある。 // そのときはRefererで見る。 const ref = req.headers.get("referer") || ""; return ALLOWED_ORIGINS.some((o) => ref.startsWith(o)); } // ========================================================= // 回数を数える // ========================================================= async function checkUsage(keyHash: string): Promise<{ allowed: boolean; remaining: number }> { const today = todayJST(); const { data: row } = await supabase .from("free_usage") .select("today_date, today_count") .eq("key_hash", keyHash) .maybeSingle(); if (!row) return { allowed: true, remaining: FREE_DAILY_LIMIT }; if (row.today_date !== today) return { allowed: true, remaining: FREE_DAILY_LIMIT }; const used = Number(row.today_count) || 0; return { allowed: used < FREE_DAILY_LIMIT, remaining: Math.max(0, FREE_DAILY_LIMIT - used) }; } async function countUp(keyHash: string): Promise { const today = todayJST(); const now = new Date().toISOString(); const { data: row } = await supabase .from("free_usage") .select("today_date, today_count, total_count") .eq("key_hash", keyHash) .maybeSingle(); if (!row) { await supabase.from("free_usage").insert({ key_hash: keyHash, today_date: today, today_count: 1, total_count: 1, first_used_at: now, last_used_at: now, }); return FREE_DAILY_LIMIT - 1; } const sameDay = row.today_date === today; const todayCount = (sameDay ? Number(row.today_count) || 0 : 0) + 1; await supabase .from("free_usage") .update({ today_date: today, today_count: todayCount, total_count: (Number(row.total_count) || 0) + 1, last_used_at: now, }) .eq("key_hash", keyHash); return Math.max(0, FREE_DAILY_LIMIT - todayCount); } // ========================================================= // 辞書に貯める（型番・メーカー名・商品名だけ） // ========================================================= async function saveToDict(product: any) { try { const asin = String(product?.asin || "").trim(); if (!asin) return; const model = String(product?.model || product?.partNumber || "").trim() || null; const brand = String(product?.brand || product?.manufacturer || "").trim() || null; const title = String(product?.title || "").trim() || null; const now = new Date().toISOString(); const { data: exist } = await supabase .from("product_dict") .select("asin, model, brand, title, lookup_count") .eq("asin", asin) .maybeSingle(); if (!exist) { await supabase.from("product_dict").insert({ asin, model, brand, title, lookup_count: 1, created_at: now, updated_at: now, }); return; } // すでにある行は、空だったところだけ埋める。上書きはしない。 const patch: Record = { lookup_count: (Number(exist.lookup_count) || 0) + 1, updated_at: now, }; if (!exist.model && model) patch.model = model; if (!exist.brand && brand) patch.brand = brand; if (!exist.title && title) patch.title = title; await supabase.from("product_dict").update(patch).eq("asin", asin); } catch (_e) { // 辞書は貯まらなくても解析は返す。ここで失敗しても黙って続ける。 } } // ========================================================= // 本体 // ========================================================= serve(async (req: Request) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: CORS_HEADERS }); } if (req.method !== "POST") { return json({ success: false, message: "POSTで呼んでください" }, 405); } if (!originAllowed(req)) { return json({ success: false, message: "この場所からは利用できません" }, 403); } let body: any = {}; try { body = await req.json(); } catch (_e) { return json({ success: false, message: "受け取れませんでした" }, 400); } // --------------------------------------------------------- // 温めるだけの呼び出し。何もせずすぐ返す。 // --------------------------------------------------------- if (body.ping === true) { return json({ success: true, pong: true }); } const keepaKey = String(body.keepaKey || "").trim(); if (!keepaKey) { return json({ success: false, needKey: true, message: "KeepaのAPIキーを入力してください" }); } const keyHash = await sha256(keepaKey); const pre = await checkUsage(keyHash); if (!pre.allowed) { return json({ success: false, limitReached: true, remaining: 0, limit: FREE_DAILY_LIMIT, message: `無料版は1日${FREE_DAILY_LIMIT}回までです。明日また使えます。`, }); } // --------------------------------------------------------- // 入力を1つ受け取って、ASINまで自分で辿り着く。 // ASINならそのまま。それ以外はKeepaの検索を1回はさむ。 // ブラウザとの往復は、どちらの場合も1回で済む。 // --------------------------------------------------------- const term = String(body.term || body.asin || "").trim(); if (!term) return json({ success: false, message: "調べるものがありません" }); const stats = String(body.stats || "90"); const buybox = body.buybox === false ? "" : "&buybox=1"; const tail = `&stats=${encodeURIComponent(stats)}${buybox}`; const isAsin = /^[A-Z0-9]{10}$/i.test(term); let asin = term; let searchTokens = 0; let kJson: any = null; let route = "asin"; // --------------------------------------------------------- // 🌟 JANを code で直接引く案は取りやめた。 // // Keepaの商品取得は code(JAN/EAN/UPC) でも引けるが、 // 1つのJANに色ちがい・容量ちがい・旧パッケージなどが // ぶら下がっていると、該当する商品を全部返してきて // 1件ずつ課金される。実測で38消費した（検索経由なら15）。 // 件数を絞る指定が無いので、平均では必ず損をする。 // 検索は1件に絞ってから取りにいくので、こちらのほうが安い。 // --------------------------------------------------------- if (!isAsin) { try { const sUrl = `https://api.keepa.com/search?key=${keepaKey}&domain=5&type=product&term=${encodeURIComponent(term)}`; const sRes = await fetch(sUrl); const sJson = await sRes.json(); searchTokens = Number(sJson?.tokensConsumed) || 0; if (!sJson?.products || sJson.products.length === 0) { return json({ success: true, notFound: true, remaining: pre.remaining, limit: FREE_DAILY_LIMIT, tokensConsumed: searchTokens, tokensLeft: sJson?.tokensLeft, route: "search", }); } asin = sJson.products[0].asin; route = "search"; } catch (e) { return json({ success: false, message: `Keepaの検索に失敗しました: ${String(e).slice(0, 200)}` }); } } // --------------------------------------------------------- // まだ商品を持っていなければ、ASINで取りにいく // --------------------------------------------------------- if (!kJson) { try { const url = `https://api.keepa.com/product?key=${keepaKey}&domain=5&asin=${encodeURIComponent(asin)}${tail}`; const kRes = await fetch(url); kJson = await kRes.json(); } catch (e) { return json({ success: false, message: `Keepaへの問い合わせに失敗しました: ${String(e).slice(0, 200)}` }); } } const prodTokens = Number(kJson?.tokensConsumed) || 0; const tokensConsumed = searchTokens + prodTokens; // Keepaが商品を返さなかったときは、回数を減らさない const product = kJson?.products?.[0]; if (!product) { return json({ success: true, notFound: true, keepa: kJson, remaining: pre.remaining, limit: FREE_DAILY_LIMIT, tokensConsumed, tokensLeft: kJson?.tokensLeft, route, }); } const remaining = await countUp(keyHash); await saveToDict(product); return json({ success: true, keepa: kJson, asin, remaining, limit: FREE_DAILY_LIMIT, tokensConsumed, tokensLeft: kJson?.tokensLeft, route, }); });
